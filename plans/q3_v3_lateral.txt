-- nhóm, nên bỏ hẳn GROUP BY đó đi. Sinh danh sách mốc tháng bằng
-- generate_series rồi với mỗi tháng cộng riêng qua LATERAL.
-- Không còn gom nhóm trên biểu thức => không còn chỗ để đoán sai
-- => không còn Sort 250.983 dòng => không còn đổ đĩa.

EXPLAIN (ANALYZE, BUFFERS)
SELECT m.month, x.revenue
FROM generate_series(
       date_trunc('month', TIMESTAMPTZ '2024-01-01'),
       (SELECT max(created_at) FROM orders),
       interval '1 month') AS m(month)
CROSS JOIN LATERAL (
  SELECT SUM(o.total) AS revenue
  FROM orders o
  WHERE o.status = 'paid'
    AND o.created_at >= m.month
    AND o.created_at <  m.month + interval '1 month'
) x
WHERE x.revenue IS NOT NULL
ORDER BY m.month;
