-- ---------- (C) Thử tạo index trên biểu thức: KHÔNG ĐƯỢC ----------
CREATE INDEX idx_orders_day ON orders (date_trunc('day', created_at));

ERROR:  functions in index expression must be marked IMMUTABLE

-- Vì sao: created_at là TIMESTAMPTZ, và date_trunc(text, timestamptz) chỉ là
-- STABLE chứ không IMMUTABLE — kết quả của nó phụ thuộc tham số TimeZone của
-- session. Cùng một dòng, hai session khác múi giờ sẽ cho hai giá trị khác
-- nhau, nên không thể lưu sẵn vào index.
--
--   SELECT proname, provolatile FROM pg_proc WHERE proname='date_trunc';
--     date_trunc(text, timestamp without time zone)    -> IMMUTABLE
--     date_trunc(text, timestamp with time zone)       -> STABLE      <-- ta đang dùng cái này
--     date_trunc(text, timestamp with time zone, text) -> IMMUTABLE   <-- có chỉ định múi giờ
