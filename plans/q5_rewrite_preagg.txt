EXPLAIN (ANALYZE, BUFFERS)
SELECT c.id, c.email, COALESCE(o.cnt,0) AS order_count
FROM customers c
LEFT JOIN (SELECT customer_id, count(*) AS cnt FROM orders GROUP BY customer_id) o
  ON o.customer_id = c.id
WHERE c.country='VN';
