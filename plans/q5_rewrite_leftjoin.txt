-- Q5 — PHƯƠNG ÁN VIẾT LẠI 1: LEFT JOIN + GROUP BY
-- Đây là lời khuyên kinh điển "correlated subquery thì viết lại thành JOIN".
-- Kết quả: CHẬM HƠN 4,5 LẦN so với bản gốc.

EXPLAIN (ANALYZE, BUFFERS)
SELECT c.id, c.email, count(o.id) AS order_count
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.id
WHERE c.country = 'VN'
GROUP BY c.id, c.email;
